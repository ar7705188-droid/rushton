    /* =========================================================================
       AUDIO SYNTHESIZER (WEB AUDIO API)
       ========================================================================= */
    class AudioSystem {
      constructor() { this.ctx = null; }
      init() {
        if (!this.ctx) this.ctx = new (window.AudioContext || window.webkitAudioContext)();
        if (this.ctx.state === 'suspended') this.ctx.resume();
      }
      playChime(isCorrect = true) {
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const freqs = isCorrect ? [523.25, 659.25, 783.99] : [180, 140, 100];
        freqs.forEach((f, i) => {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = isCorrect ? 'sine' : 'sawtooth';
          osc.frequency.setValueAtTime(f, now + i * 0.08);
          gain.gain.setValueAtTime(0.3, now + i * 0.08);
          gain.gain.exponentialRampToValueAtTime(0.01, now + i * 0.08 + 0.2);
          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start(now + i * 0.08);
          osc.stop(now + i * 0.08 + 0.2);
        });
      }
      playBsoda() {
        if (!this.ctx) return;
        const bufferSize = this.ctx.sampleRate * 0.4;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) output[i] = Math.random() * 2 - 1;
        const whiteNoise = this.ctx.createBufferSource();
        whiteNoise.buffer = buffer;
        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(1200, this.ctx.currentTime);
        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(0.6, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.4);
        whiteNoise.connect(filter); filter.connect(gain); gain.connect(this.ctx.destination);
        whiteNoise.start();
      }
      playJumpScare() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(90, this.ctx.currentTime);
        osc.frequency.linearRampToValueAtTime(700, this.ctx.currentTime + 0.35);
        gain.gain.setValueAtTime(1.0, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 1.2);
        osc.connect(gain); gain.connect(this.ctx.destination);
        osc.start(); osc.stop(this.ctx.currentTime + 1.2);
      }
    }

    function createWallTexture(baseColor = '#fef08a', trimColor = '#1d4ed8') {
      const canvas = document.createElement('canvas');
      canvas.width = 512; canvas.height = 512;
      const ctx = canvas.getContext('2d');
      
      // Base Wall Paint Texture
      ctx.fillStyle = baseColor;
      ctx.fillRect(0, 0, 512, 512);

      // Brick pattern overlays
      ctx.strokeStyle = 'rgba(0, 0, 0, 0.08)';
      ctx.lineWidth = 3;
      const rowHeight = 32;
      const brickWidth = 64;
      for (let y = 0; y < 512; y += rowHeight) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(512, y); ctx.stroke();
        const offset = (y / rowHeight) % 2 === 0 ? 0 : brickWidth / 2;
        for (let x = offset; x < 512; x += brickWidth) {
          ctx.beginPath(); ctx.moveTo(x, y); ctx.lineTo(x, y + rowHeight); ctx.stroke();
        }
      }

      // Baseboard Trim & Top Molding
      ctx.fillStyle = trimColor;
      ctx.fillRect(0, 0, 512, 40);
      ctx.fillRect(0, 460, 512, 52);
      ctx.fillStyle = 'rgba(0, 0, 0, 0.25)';
      ctx.fillRect(0, 36, 512, 4);
      ctx.fillRect(0, 460, 512, 6);

      const texture = new THREE.CanvasTexture(canvas);
      texture.anisotropy = 8;
      return texture;
    }

    function createRealisticFloorTexture() {
      const canvas = document.createElement('canvas');
      canvas.width = 512; canvas.height = 512;
      const ctx = canvas.getContext('2d');

      // Linoleum Tile Floor Grid
      ctx.fillStyle = '#e2e8f0'; ctx.fillRect(0, 0, 512, 512);
      ctx.fillStyle = '#cbd5e1';
      for (let i = 0; i < 512; i += 128) {
        for (let j = 0; j < 512; j += 128) {
          if ((i + j) % 256 === 0) ctx.fillRect(i, j, 128, 128);
        }
      }

      // Tile Grout & Dirt Spots
      ctx.strokeStyle = '#64748b'; ctx.lineWidth = 4;
      for (let k = 0; k <= 512; k += 128) {
        ctx.beginPath(); ctx.moveTo(k, 0); ctx.lineTo(k, 512); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(0, k); ctx.lineTo(512, k); ctx.stroke();
      }

      const texture = new THREE.CanvasTexture(canvas);
      texture.wrapS = THREE.RepeatWrapping;
      texture.wrapT = THREE.RepeatWrapping;
      texture.repeat.set(12, 12);
      return texture;
    }

      texture.wrapS = THREE.RepeatWrapping;

      return texture;
    }

    function buildSchoolMap() {
      const wallGeo = new THREE.BoxGeometry(CELL_SIZE, 3.6, CELL_SIZE);
      
      // Upgrade materials to Standard Materials for realistic lighting & shadows
      const hallwayWallMat = new THREE.MeshStandardMaterial({ map: createWallTexture('#fef08a', '#2563eb'), roughness: 0.6, metalness: 0.1 });
      const classWallMat = new THREE.MeshStandardMaterial({ map: createWallTexture('#bae6fd', '#0284c7'), roughness: 0.6, metalness: 0.1 });
      const cafeteriaWallMat = new THREE.MeshStandardMaterial({ map: createWallTexture('#fca5a5', '#dc2626'), roughness: 0.5, metalness: 0.2 });
      const exitDoorMat = new THREE.MeshStandardMaterial({ map: createWallTexture('#ef4444', '#15803d'), roughness: 0.4, metalness: 0.3 });

      // Realistic Floor Plane
      const floorGeo = new THREE.PlaneGeometry(MAP_SIZE * CELL_SIZE, MAP_SIZE * CELL_SIZE);
      const floorMat = new THREE.MeshStandardMaterial({ map: createRealisticFloorTexture(), roughness: 0.3, metalness: 0.1 });
      const floorMesh = new THREE.Mesh(floorGeo, floorMat);
      floorMesh.rotation.x = -Math.PI / 2;
      floorMesh.position.set((MAP_SIZE * CELL_SIZE)/2 - CELL_SIZE/2, 0, (MAP_SIZE * CELL_SIZE)/2 - CELL_SIZE/2);
      scene.add(floorMesh);

      // Realistic Ceiling Plane
      const ceilGeo = new THREE.PlaneGeometry(MAP_SIZE * CELL_SIZE, MAP_SIZE * CELL_SIZE);
      const ceilMat = new THREE.MeshStandardMaterial({ map: createRealisticCeilingTexture(), roughness: 0.9, metalness: 0.0 });
      const ceilMesh = new THREE.Mesh(ceilGeo, ceilMat);
      ceilMesh.rotation.x = Math.PI / 2;
      ceilMesh.position.set((MAP_SIZE * CELL_SIZE)/2 - CELL_SIZE/2, 3.6, (MAP_SIZE * CELL_SIZE)/2 - CELL_SIZE/2);
      scene.add(ceilMesh);

      // Build wall boxes for all grid tiles marked as walls or doors
      for (let r = 0; r < MAP_SIZE; r++) {
        for (let c = 0; c < MAP_SIZE; c++) {
          const tile = mapGrid[r][c];
          const posX = c * CELL_SIZE;
          const posZ = r * CELL_SIZE;

          if (tile === 1) {
            const isCafeteria = (r > 0 && mapGrid[r-1][c] === 5) || (r < MAP_SIZE-1 && mapGrid[r+1][c] === 5) ||
                                (c > 0 && mapGrid[r][c-1] === 5) || (c < MAP_SIZE-1 && mapGrid[r][c+1] === 5);
            const isClass = (r > 0 && mapGrid[r-1][c] === 2) || (r < MAP_SIZE-1 && mapGrid[r+1][c] === 2) ||
                            (c > 0 && mapGrid[r][c-1] === 2) || (c < MAP_SIZE-1 && mapGrid[r][c+1] === 2);
            
            const wallMat = isCafeteria ? cafeteriaWallMat : (isClass ? classWallMat : hallwayWallMat);
            const wall = new THREE.Mesh(wallGeo, wallMat);
            wall.position.set(posX, 1.8, posZ);
            scene.add(wall);
          } else if (tile === 4) {
            const door = new THREE.Mesh(wallGeo, exitDoorMat);
            door.position.set(posX, 1.8, posZ);
            scene.add(door);
          } else if (tile === 3) {
            GAME.player = { x: posX, z: posZ, radius: 1.8 };
            camera.position.set(posX, 1.6, posZ);
          } else if (tile === 5) {
            if (r % 2 === 0 && c % 2 === 0) {
              const tableGeo = new THREE.BoxGeometry(2.4, 1.0, 1.4);
              const tableMat = new THREE.MeshStandardMaterial({ map: createCafeteriaTableTexture(), roughness: 0.4 });
              const table = new THREE.Mesh(tableGeo, tableMat);
              table.position.set(posX, 0.5, posZ);
              scene.add(table);
            }
          }

          // Overhead Realistic Fluorescent Light Fixtures
          if ((tile === 0 || tile === 5) && (r % 3 === 0) && (c % 3 === 0)) {
            const lightColor = tile === 5 ? 0xffedd5 : 0xf8fafc;
            const light = new THREE.PointLight(lightColor, 1.2, 16);
            light.position.set(posX, 3.2, posZ);
            scene.add(light);
          }
        }
      }
    }

    function init() {
      scene = new THREE.Scene();
      scene.background = new THREE.Color(0x020617);
      scene.fog = new THREE.FogExp2(0x020617, 0.03);

      camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 100);
      camera.rotation.order = 'YXZ';

      renderer = new THREE.WebGLRenderer({ antialias: true });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.toneMapping = THREE.ACESFilmicToneMapping;
      renderer.toneMappingExposure = 1.1;
      document.getElementById('game-container').appendChild(renderer.domElement);

      const ambientLight = new THREE.AmbientLight(0xffffff, 0.35);
      scene.add(ambientLight);

      // Realistic Focused Flashlight
      flashlight = new THREE.SpotLight(0xfffbeb, 3.5, 25, Math.PI / 6, 0.4, 1.2);
      camera.add(flashlight);
      camera.add(flashlight.target);
      flashlight.target.position.set(0, 0, -1);
      scene.add(camera);

      buildSchoolMap();
      spawnEntities();
      setupEventListeners();
      animate();
    }

    function setupEventListeners() {
      window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
      });

      document.addEventListener('keydown', (e) => {
        keys[e.code] = true;
        if (e.code === 'KeyF' && flashlight) flashlight.visible = !flashlight.visible;
        if (e.code === 'KeyE') checkInteractions();
        if (e.code === 'Space') usePropel();
      });

      document.addEventListener('keyup', (e) => { keys[e.code] = false; });

      const container = document.getElementById('game-container');
      container.addEventListener('click', () => {
        if (GAME.isStarted && !GAME.terminalActive) {
          container.requestPointerLock();
          checkInteractions();
        }
      });

      document.addEventListener('mousemove', (e) => {
        if (!GAME.isStarted || GAME.terminalActive || document.pointerLockElement !== container) return;
        yaw -= e.movementX * 0.0025;
        pitch -= e.movementY * 0.0025;
        pitch = Math.max(-1.4, Math.min(1.4, pitch));
        camera.rotation.y = yaw;
        camera.rotation.x = pitch;
      });

      // UI Buttons
      document.getElementById('start-btn').addEventListener('click', startGame);
      document.getElementById('restart-btn').addEventListener('click', resetGame);
      document.getElementById('victory-restart-btn').addEventListener('click', resetGame);
      document.getElementById('fullscreen-btn').addEventListener('click', toggleFullscreen);
      document.getElementById('char-dialogue-close').addEventListener('click', () => {
        document.getElementById('char-dialogue-modal').classList.add('hidden');
        GAME.terminalActive = false;
        if (document.getElementById('game-container').requestPointerLock) {
          document.getElementById('game-container').requestPointerLock();
        }
      });

      document.getElementById('math-form').addEventListener('submit', (e) => {
        e.preventDefault();
        submitMathAnswer();
      });
    }

    function toggleFullscreen() {
      if (!document.fullscreenElement) {
        document.documentElement.requestFullscreen().catch(() => {});
      } else if (document.exitFullscreen) {
        document.exitFullscreen();
      }
    }

    /* =========================================================================
       INTERACTION & ITEM LOGIC
       ========================================================================= */
    function checkInteractions() {
      const raycaster = new THREE.Raycaster();
      raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
      const intersects = raycaster.intersectObjects(interactiveObjects);

      if (intersects.length > 0 && intersects[0].distance < 3.8) {
        const obj = intersects[0].object;
        const type = obj.userData.type;

        if (type === 'notebook') {
          openMathTerminal(obj);
        } else if (type === 'coin') {
          GAME.coins++;
          document.getElementById('coin-count').innerText = GAME.coins;
          audio.playChime(true);
          scene.remove(obj);
          interactiveObjects.splice(interactiveObjects.indexOf(obj), 1);
          showMessage("+1 Coin Collected!");
        } else if (type === 'propel') {
          addInventory('Propel');
          scene.remove(obj);
          interactiveObjects.splice(interactiveObjects.indexOf(obj), 1);
        } else if (type === 'vending') {
          if (GAME.coins >= 1) {
            GAME.coins--;
            document.getElementById('coin-count').innerText = GAME.coins;
            addInventory('Propel');
            audio.playBsoda();
            showMessage("Bought Propel drink for 1 Coin!");
          } else {
            audio.playChime(false);
            showMessage("Need 1 Coin to buy Propel!");
          }
        } else if (['zyker', 'nono', 'kiki', 'myles', 'tom'].includes(type)) {
          interactWithCharacter(type);
        }
      }
    }

    function interactWithCharacter(type) {
      GAME.terminalActive = true;
      if (document.exitPointerLock) document.exitPointerLock();

      const modal = document.getElementById('char-dialogue-modal');
      const nameEl = document.getElementById('char-name');
      const textEl = document.getElementById('char-text');

      if (type === 'zyker') {
        nameEl.innerText = "Zyker (Blonde Hair)";
        textEl.innerText = '"Hey! I saw Mr. Rushton patrolling by classroom 3. Watch out for his wooden ruler!"';
        audio.playChime(true);
      } else if (type === 'nono') {
        nameEl.innerText = "Nono (Elderly Hallway Monitor)";
        textEl.innerText = '"Back in my day we had 10 notebooks! Here, take some extra stamina kid!"';
        GAME.stamina = 100;
        audio.playChime(true);
      } else if (type === 'kiki') {
        nameEl.innerText = "Kiki (Brown Hair)";
        textEl.innerText = '"Nobody likes me... but don\'t get caught by Mr. Rushton or you\'ll get a Red Sheet!"';
        audio.playChime(false);
      } else if (type === 'myles') {
        nameEl.innerText = "Myles Ori (Coolest Guy in School)";
        textEl.innerText = '"Stay cool! Here\'s a quarter for the Propel vending machine!"';
        GAME.coins += 1;
        document.getElementById('coin-count').innerText = GAME.coins;
        audio.playChime(true);
      } else if (type === 'tom') {
        nameEl.innerText = "Tom";
        textEl.innerText = '"Mmm... chocolate! Eating sweets gives me so much energy! Keep up the stamina!"';
        GAME.stamina = Math.min(100, GAME.stamina + 50);
        audio.playChime(true);
      }

      modal.classList.remove('hidden');
    }

    function addInventory(item) {
      for (let i = 0; i < 3; i++) {
        if (!GAME.inventory[i]) {
          GAME.inventory[i] = item;
          document.getElementById(`item-${i}`).innerText = '⚡';
          audio.playChime(true);
          showMessage("Picked up Propel drink!");
          return;
        }
      }
      showMessage("Inventory Full!");
    }

    function usePropel() {
      const index = GAME.inventory.indexOf('Propel');
      if (index !== -1) {
        GAME.inventory[index] = null;
        document.getElementById(`item-${index}`).innerText = '';
        GAME.stamina = 100;
        audio.playBsoda();
        showMessage("Drank Propel! Stamina Restored!");
      }
    }

    /* =========================================================================
       MATH TERMINAL LOGIC & SECRET CODE 67
       ========================================================================= */
    function openMathTerminal(obj) {
      GAME.terminalActive = true;
      GAME.currentNotebookObj = obj;
      GAME.questionIndex = 0;
      if (document.exitPointerLock) document.exitPointerLock();
      document.getElementById('math-terminal').classList.remove('hidden');
      nextMathQuestion();
    }

    function nextMathQuestion() {
      const input = document.getElementById('math-input');
      input.value = '';
      input.focus();
      GAME.q1 = Math.floor(Math.random() * 10);
      GAME.q2 = Math.floor(Math.random() * 10);
      document.getElementById('question-number').innerText = `QUESTION ${GAME.questionIndex + 1} OF 3`;
      document.getElementById('question-text').innerText = `${GAME.q1} + ${GAME.q2} = ?`;
    }

    function submitMathAnswer() {
      const val = document.getElementById('math-input').value.trim();
      
      // Secret Code 67 Trigger
      if (val === '67') {
        document.getElementById('math-terminal').classList.add('hidden');
        GAME.terminalActive = false;
        triggerRedSheet("Entered Secret Code 67!");
        return;
      }

      const isCorrect = parseInt(val) === (GAME.q1 + GAME.q2);
      audio.playChime(isCorrect);

      if (!isCorrect) {
        GAME.rushtonSpeed += 0.025; // Speed increases even faster on incorrect answers
        showMessage("Wrong answer! Mr. Rushton is speeding up!");
      }

      GAME.questionIndex++;
      if (GAME.questionIndex >= 3) {
        document.getElementById('math-terminal').classList.add('hidden');
        GAME.terminalActive = false;
        if (document.getElementById('game-container').requestPointerLock) {
          document.getElementById('game-container').requestPointerLock();
        }

        if (GAME.currentNotebookObj) {
          scene.remove(GAME.currentNotebookObj);
          interactiveObjects.splice(interactiveObjects.indexOf(GAME.currentNotebookObj), 1);
          GAME.notebooksCollected++;
          document.getElementById('notebook-count').innerText = `${GAME.notebooksCollected}/${GAME.totalNotebooks}`;
          showMessage(`Collected Notebook ${GAME.notebooksCollected}/7!`);

          if (GAME.notebooksCollected >= GAME.totalNotebooks) {
            document.getElementById('victory-modal').classList.remove('hidden');
            if (document.exitPointerLock) document.exitPointerLock();
          }
        }
      } else {
        nextMathQuestion();
      }
    }

    /* =========================================================================
       DETENTION & GAME OVER
       ========================================================================= */
    function triggerRedSheet(reason) {
      audio.playJumpScare();
      document.getElementById('detention-reason').innerText = reason;
      document.getElementById('red-sheet-modal').classList.remove('hidden');
      if (document.exitPointerLock) document.exitPointerLock();
      GAME.isStarted = false;
    }

    function resetGame() {
      location.reload();
    }

    /* =========================================================================
       GAME LOOP & MOVEMENT
       ========================================================================= */
    function animate() {
      requestAnimationFrame(animate);

      if (GAME.isStarted && !GAME.terminalActive) {
        // Boosted player movement and sprinting speeds
        const moveSpeed = (keys['ShiftLeft'] || keys['ShiftRight']) && GAME.stamina > 0 ? 0.16 : 0.095;
        
        if ((keys['ShiftLeft'] || keys['ShiftRight']) && GAME.stamina > 0) {
          GAME.stamina = Math.max(0, GAME.stamina - 0.3);
        } else if (GAME.stamina < 100) {
          GAME.stamina = Math.min(100, GAME.stamina + 0.15);
        }

        document.getElementById('stamina-bar').style.width = `${GAME.stamina}%`;

        const moveDir = new THREE.Vector3();
        if (keys['KeyW']) moveDir.z -= 1;
        if (keys['KeyS']) moveDir.z += 1;
        if (keys['KeyA']) moveDir.x -= 1;
        if (keys['KeyD']) moveDir.x += 1;

        moveDir.normalize();
        moveDir.applyAxisAngle(new THREE.Vector3(0, 1, 0), yaw);

        const nextX = camera.position.x + moveDir.x * moveSpeed;
        const nextZ = camera.position.z + moveDir.z * moveSpeed;

        // Grid Collision Checks
        const gridX = Math.floor((nextX + CELL_SIZE/2) / CELL_SIZE);
        const gridZ = Math.floor((nextZ + CELL_SIZE/2) / CELL_SIZE);

        if (gridZ >= 0 && gridZ < MAP_SIZE && gridX >= 0 && gridX < MAP_SIZE) {
          if (mapGrid[gridZ][gridX] !== 1) {
            camera.position.x = nextX;
            camera.position.z = nextZ;
          }
        }

        // Update Antagonist Movement (Mr. Rushton chases player)
        characters.forEach(char => {
          if (char.userData.type === 'rushton') {
            const dir = new THREE.Vector3().subVectors(camera.position, char.position);
            dir.y = 0;
            const dist = dir.length();
            if (dist < 1.8) {
              triggerRedSheet("Caught by Mr. Rushton!");
            } else {
              dir.normalize();
              char.position.addScaledVector(dir, GAME.rushtonSpeed);
            }
          }
        });
      }

      renderer.render(scene, camera);
    }

    window.onload = init;
